    bf16* dx;       // [B, N, d_model]   BWD ONLY — grad to prev layer
};

struct SavedTensors {                    // FWD writes, BWD reads. Inference skips.
    bf16* O;        // [B, N, H, d_h]    attention output (also used in O-LoRA bwd)
    fp32* LSE;      // [B, H, N]         logsumexp, σ_h folded into denominator
};

struct Weights {                         // Read-only in fwd AND bwd.
    // Q path
    fp8*  Aq;             // [d_model, r]
    fp8*  Bq;             // [r, H*d_h]

    // Shared single-head KV (the "GQA-with-1-head" trick)
    fp8*  Wkv;            // [d_model, 2*d_h]

    // Compressor bundle — the only custom learnable group
    fp8*  Wcomp_kv;       // [d_model, 2*d_h]   inner KV proj
    fp32* w_score;        // [d_h]              per-token scoring vector
    fp32* ape;            // [m', d_h]          intra-block positional bias
    fp32* norm_inner;     // [d_h]              RMSNorm inside compressor

    // Outer attention pre-norms + sink
    fp32* normQ;          // [d_h]
    fp32* normKc;         // [d_h]
    fp32* sink;           // [H]                σ_h, per-head learnable

    // O path (grouped, 8 groups)
    fp8*  Ao;             // [H*d_h, r]
    fp8*  Bo;             // [r, d_model]

    WeightScales scales;  // e8m0fnu block scales, one per fp8 weight
};

struct Gradients {                       // BWD ONLY. All fp32 master.
    fp32 *dAq, *dBq, *dWkv;
    fp32 *dWcomp_kv, *dw_score, *dape, *dnorm_inner;
    fp32 *dnormQ, *dnormKc, *dsink;     // dsink: [H]
    fp32 *dAo, *dBo;
};

struct CacheState {                      // DECODE ONLY. Absent in prefill & training.
    fp8*  Kc;             // [B, M_max, d_h]   compressed K, append-only
    fp8*  Vc;             // [B, M_max, d_h]   compressed V, append-only
    bf16* K_swa_ring;     // [B, W, d_h]       last-128 raw K
    bf16* V_swa_ring;     // [B, W, d_h]
    bf16* K_partial;      // [B, m', d_h]      in-flight uncompressed block
    bf16* V_partial;      // [B, m', d_h]
    int   partial_count;  //                   0..127
    int   M_cur;          //                   current compressed length
    int   pos;            //                   global token position
};
